% ──────────────────────────────────────────────────────────────
% \letterhead{badge_nr}  — nagłówek ANS z poziomą linią
% ──────────────────────────────────────────────────────────────
\newcommand{\letterhead}[1]{%
  \noindent
  \begin{minipage}[t]{0.68\linewidth}
    {\bfseries\fontsize{10}{13}\selectfont
      Akademia Nauk Stosowanych w~Elblągu}\\[3pt]
    {\itshape\fontsize{9}{12}\selectfont\color{darkgray}%
      Instytut Informatyki Stosowanej im.~Krzysztofa Brzeskiego}
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.30\linewidth}
    \raggedleft
    \setlength{\fboxsep}{4pt}
    \setlength{\fboxrule}{0.75pt}
    {\color{darkgray}\fbox{%
      \parbox{0.92\linewidth}{%
        \centering
        \fontsize{7.5}{9}\selectfont\bfseries
        Zał.~nr~#1%
      }%
    }}%
  \end{minipage}
  \par\vspace{4pt}
  \noindent{\color{sepgray}\rule{\linewidth}{1.5pt}}
  \par\vspace{6pt}
}

% ──────────────────────────────────────────────────────────────
% \pfield{label}{value}  — pole z etykietą i podkreśleniem
% ──────────────────────────────────────────────────────────────
\newcommand{\pfield}[2]{%
  \par
  {\fontsize{8}{10}\selectfont\sffamily\color{labelgray}#1}
  \par\noindent
  {\raggedright\fontsize{10}{13}\selectfont #2\strut\par}%
  \par\vspace{-2pt}
  \noindent{\color{midgray}\rule{\linewidth}{0.5pt}}
  \par\vspace{3pt}
}

% ──────────────────────────────────────────────────────────────
% \psig{label}{value}  — pole podpisu z przerywaną ramką
% ──────────────────────────────────────────────────────────────
\newcommand{\psig}[2]{%
  \par\vspace{2pt}
  \begin{tcolorbox}[
    enhanced,
    colback=white,
    frame hidden,
    borderline={0.4pt}{0pt}{midgray, dashed},
    arc=0pt, outer arc=0pt,
    left=5pt, right=5pt, top=5pt, bottom=10pt,
    boxsep=0pt
  ]
  {\fontsize{8}{10}\selectfont\sffamily\itshape\color{labelgray}#1}\\[4pt]
  {\fontsize{10}{13}\selectfont #2\strut}
  \vspace{6pt}
  \end{tcolorbox}%
  \par\vspace{4pt}
}

% ──────────────────────────────────────────────────────────────
% \sectionlabel{text}  — nagłówek sekcji (uppercase, letter-spacing)
% ──────────────────────────────────────────────────────────────
\newcommand{\sectionlabel}[1]{%
  \par\vspace{4pt}
  {\fontsize{7.5}{10}\selectfont\sffamily\bfseries\color{darkgray}%
   \addfontfeature{LetterSpace=10}\MakeUppercase{#1}}%
  \par\vspace{3pt}
}

% ──────────────────────────────────────────────────────────────
% \doctitle{text}  — wyśrodkowany tytuł dokumentu
% ──────────────────────────────────────────────────────────────
\newcommand{\doctitle}[1]{%

    {\fontsize{12}{15}\selectfont\bfseries#1}
  \end{center}
  \vspace{2pt}
}
