% ──────────────────────────────────────────────────────────────
% \letterhead{badge_nr}  — ANS header with horizontal rule
% ──────────────────────────────────────────────────────────────
\newcommand{\letterhead}[1]{%
  \noindent
  \begin{minipage}[t]{0.68\linewidth}
    {\bfseries\fontsize{10}{13}\selectfont
      Akademia Nauk Stosowanych w~Elblągu}\\[3pt]
    {\itshape\fontsize{9}{12}\selectfont\color{darkgray}%
      Instytut Informatyki Stosowanej im.~Krzysztofa Brzeskiego}
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.30\linewidth}
    \raggedleft
    \setlength{\fboxsep}{4pt}
    \fbox{%
      \parbox{0.92\linewidth}{%
        \centering
        \fontsize{7.5}{9}\selectfont\bfseries
        Zał. nr~#1%
      }%
    }%
  \end{minipage}
  \par\vspace{4pt}
  \noindent\rule{\linewidth}{1pt}
  \par\vspace{6pt}
}

% ──────────────────────────────────────────────────────────────
% \pfield{label}{value}  — labeled field with underline
% ──────────────────────────────────────────────────────────────
\newcommand{\pfield}[2]{%
  \par
  {\fontsize{8}{10}\selectfont\sffamily\color{sectiongray}#1}
  \par\noindent
  \makebox[\linewidth][l]{\fontsize{10}{13}\selectfont #2}%
  \par\vspace{-2pt}
  \noindent{\color{midgray}\rule{\linewidth}{0.4pt}}
  \par\vspace{3pt}
}

% ──────────────────────────────────────────────────────────────
% \psig{label}{value}  — signature box with dashed border
%  Uses \fbox with a minimum height achieved via \vspace inside
% ──────────────────────────────────────────────────────────────
\newcommand{\psig}[2]{%
  \par
  \begingroup
  \setlength{\fboxsep}{5pt}
  \setlength{\fboxrule}{0.4pt}
  \noindent\fbox{%
    \begin{minipage}[c][36pt][t]{\dimexpr\linewidth-\fboxsep*2-\fboxrule*2\relax}
      {\fontsize{8}{10}\selectfont\sffamily\itshape\color{sectiongray}#1}\\[4pt]
      {\fontsize{10}{13}\selectfont #2}
    \end{minipage}%
  }%
  \endgroup
  \par\vspace{4pt}
}

% ──────────────────────────────────────────────────────────────
% \sectionlabel{text}  — small uppercase section heading
% ──────────────────────────────────────────────────────────────
\newcommand{\sectionlabel}[1]{%
  \par\vspace{4pt}
  {\fontsize{7.5}{10}\selectfont\sffamily\bfseries\color{sectiongray}#1}%
  \par\vspace{3pt}
}

% ──────────────────────────────────────────────────────────────
% \doctitle{text}  — centered bold uppercase title
% ──────────────────────────────────────────────────────────────
\newcommand{\doctitle}[1]{%
  \begin{center}
    {\fontsize{12}{15}\selectfont\bfseries #1}
  \end{center}
  \vspace{2pt}
}

% ──────────────────────────────────────────────────────────────
% \docsubtitle{text}  — centered italic subtitle
% ──────────────────────────────────────────────────────────────
\newcommand{\docsubtitle}[1]{%
  \begin{center}
    {\fontsize{10}{13}\selectfont\itshape\color{darkgray}#1}
  \end{center}
  \vspace{4pt}
}

% ──────────────────────────────────────────────────────────────
% \sectionrule  — thin horizontal separator between sections
% ──────────────────────────────────────────────────────────────
\newcommand{\sectionrule}{%
  \par\vspace{4pt}
  \noindent{\color{lightgray}\rule{\linewidth}{0.5pt}}%
  \par\vspace{3pt}
}

% ──────────────────────────────────────────────────────────────

  }%
